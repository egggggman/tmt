class Pipe:
    def __init__(self,shared):self.shared=shared
    def send_bytes(self,data):self.shared['data']=data
    def close(self):pass
    def poll(self,_timeout):return True
    def recv_bytes(self,_maximum):
        if self.shared.get('delay'):time.sleep(self.shared['delay'])
        return self.shared.get('data',b'')

class ImmediateProcess:
    pid=123
    def __init__(self,target,args):self.target=target;self.args=args
    def start(self):self.target(*self.args)
    def is_alive(self):return False
    def join(self):pass

class ImmediateContext:
    def __init__(self,shared):self.shared=shared
    def Pipe(self,**_):return Pipe(self.shared),Pipe(self.shared)
    def Process(self,target,args,**_):return ImmediateProcess(target,args)

def native_reader(path,sharing):
    import ctypes
    import ctypes.wintypes as w
    kernel=ctypes.WinDLL('kernel32',use_last_error=True)
    kernel.CreateFileW.argtypes=[w.LPCWSTR,w.DWORD,w.DWORD,w.LPVOID,w.DWORD,w.DWORD,w.HANDLE]
    kernel.CreateFileW.restype=w.HANDLE;kernel.CloseHandle.argtypes=[w.HANDLE]
    handle=kernel.CreateFileW(str(path),0x80000000,sharing,None,3,0x80,None)
    assert handle!=w.HANDLE(-1).value
    return kernel,handle
